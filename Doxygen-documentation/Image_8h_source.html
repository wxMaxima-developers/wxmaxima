202 void ClearCache()
203 {
204 if(m_loadImageTask)
205 m_loadImageTask->Wait();
206 // Querying the size of an invalid bitmap trips a wxWidgets assertion (and,
207 // under a GUI assert handler, hangs). A not-yet-rendered cache is already
208 // "clear", so only shrink a bitmap that actually exists.
209 if (m_scaledBitmap.IsOk() &&
210 ((m_scaledBitmap.GetWidth() > 1) || (m_scaledBitmap.GetHeight() > 1)))
211 m_scaledBitmap.Create(1, 1);
212 // An SVG bundle holds the parsed document plus the last bitmap it
213 // rendered, at full size. SvgBundle() makes a new one when needed.
214 m_svgBundle = wxBitmapBundle();
215 }
