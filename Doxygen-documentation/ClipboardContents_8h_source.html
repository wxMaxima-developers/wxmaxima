79template <class T> class LazyValue
80{
81public:
82 LazyValue() = default;
83 explicit LazyValue(std::function<T()> renderer) : m_renderer(std::move(renderer)) {}
84
86 bool IsOffered() const { return m_value.has_value() || static_cast<bool>(m_renderer); }
88 bool IsRendered() const { return m_value.has_value(); }
89
91 const T &Get() const {
92 if (!m_value) {
93 if (m_renderer)
94 m_value = m_renderer();
95 else
96 m_value = T();
97 m_renderer = nullptr;
98 }
99 return *m_value;
100 }
101
102private:
103 mutable std::function<T()> m_renderer;
104 mutable std::optional<T> m_value;
105};
