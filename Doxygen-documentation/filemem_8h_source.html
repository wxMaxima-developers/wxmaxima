43struct filemem {
44 FILE *fp;
45 unsigned char *mem;
46 size_t memsize; /* Size of `mem` buffer (capacity) */
47 size_t mempos; /* Current position */
48 size_t memend; /* For use by `zint_fm_seek()`, points to highest `mempos` reached */
49 int flags; /* BARCODE_MEMORY_FILE or BARCODE_STDOUT */
50 int err; /* `errno` values, reset only on `zint_fm_open()` */
51#ifdef Z_NO_VSNPRINTF
52 FILE *fp_null; /* Only used for BARCODE_MEMORY_FILE */
53#endif
54};
