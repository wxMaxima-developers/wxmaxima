50 /* Vector elements - see vector header `zint_vector` below */
51 struct zint_vector_rect {
52 float x, y; /* Top left */
53 float height, width;
54 int colour; /* -1 for foreground, 1-8 for Cyan, Blue, Magenta, Red, Yellow, Green, Black, White */
55 struct zint_vector_rect *next; /* Pointer to next rectangle */
56 };
57
58 struct zint_vector_hexagon {
59 float x, y; /* Centre */
60 float diameter; /* Short (minimal) diameter (i.e. diameter of inscribed circle) */
61 int rotation; /* 0, 90, 180, 270 degrees, where 0 has apex at top, i.e. short diameter is horizontal */
62 struct zint_vector_hexagon *next; /* Pointer to next hexagon */
63 };
64
65 struct zint_vector_string {
66 float x, y; /* x is relative to halign (i.e. centre, left, right), y is relative to baseline */
67 float fsize; /* Font size */
68 float width; /* Rendered width estimate */
69 int length; /* Number of characters (bytes) */
70 int rotation; /* 0, 90, 180, 270 degrees */
71 int halign; /* Horizontal alignment: 0 for centre, 1 for left, 2 for right (end) */
72 unsigned char *text; /* UTF-8, NUL-terminated */
73 struct zint_vector_string *next; /* Pointer to next string */
74 };
75
76 struct zint_vector_circle {
77 float x, y; /* Centre */
78 float diameter; /* Circle diameter. Does not include width (if any) */
79 float width; /* Width of circle perimeter (circumference). 0 for fill (disc) */
80 int colour; /* Zero for draw with foreground colour (else draw with background colour (legacy)) */
81 struct zint_vector_circle *next; /* Pointer to next circle */
82 };
83
84 /* Vector header */
85 struct zint_vector {
86 float width, height; /* Width, height of barcode image (including text, whitespace) */
87 struct zint_vector_rect *rectangles; /* Pointer to first rectangle */
88 struct zint_vector_hexagon *hexagons; /* Pointer to first hexagon */
89 struct zint_vector_string *strings; /* Pointer to first string */
90 struct zint_vector_circle *circles; /* Pointer to first circle */
91 };
92
93 /* Structured Append info (see `symbol->structapp` below) - ignored unless `zint_structapp.count` is non-zero */
94 struct zint_structapp {
95 int index; /* Position in Structured Append sequence, 1-based. Must be <= `count` */
96 int count; /* Number of symbols in Structured Append sequence. Set >= 2 to add SA Info */
97 char id[32]; /* Optional ID to distinguish sequence, ASCII, NUL-terminated unless max 32 long */
98 };
99
100 /* Segment for use with API `ZBarcode_Encode_Segs()` and `content_segs` */
101 struct zint_seg {
102 unsigned char *source; /* Data to encode */
103 int length; /* Length of `source`. If 0 or negative, `source` must be NUL-terminated */
104 int eci; /* Extended Channel Interpretation */
105 };
106
107 /* Main symbol structure */
108 struct zint_symbol {
109 int symbology; /* Symbol to use (see BARCODE_XXX below) */
110 float height; /* Barcode height in X-dimensions (ignored for fixed-width barcodes) */
111 float scale; /* Scale factor when printing barcode, i.e. adjusts X-dimension. Default 1 */
112 int whitespace_width; /* Width in X-dimensions of whitespace to left & right of barcode */
113 int whitespace_height; /* Height in X-dimensions of whitespace above & below the barcode */
114 int border_width; /* Size of border in X-dimensions */
115 int output_options; /* Various output parameters (bind, box etc, see below) */
116 char fgcolour[16]; /* Foreground as hexadecimal RGB/RGBA or decimal "C,M,Y,K" string, NUL-terminated */
117 char bgcolour[16]; /* Background as hexadecimal RGB/RGBA or decimal "C,M,Y,K" string, NUL-terminated */
118 char *fgcolor; /* Pointer to fgcolour (alternate spelling) */
119 char *bgcolor; /* Pointer to bgcolour (alternate spelling) */
120 char outfile[256]; /* Name of file to output to, NUL-terminated. Default "out.png" ("out.gif" if no PNG) */
121 char primary[128]; /* Primary message data (MaxiCode, Composite), NUL-terminated */
122 int option_1; /* Symbol-specific options (see "../docs/manual.txt") */
123 int option_2; /* Symbol-specific options */
124 int option_3; /* Symbol-specific options */
125 int show_hrt; /* Show (1) or hide (0) Human Readable Text (HRT). Default 1 */
126 int input_mode; /* Encoding of input data (see DATA_MODE etc below). Default DATA_MODE */
127 int eci; /* Extended Channel Interpretation. Default 0 (none) */
128 float dpmm; /* Resolution of output in dots per mm (BMP/EMF/PCX/PNG/TIF only). Default 0 (none) */
129 float dot_size; /* Size of dots used in BARCODE_DOTTY_MODE. Default 0.8 */
130 float text_gap; /* Gap between barcode and text (HRT) in X-dimensions. Default 1 */
131 float guard_descent; /* Height in X-dimensions that EAN/UPC guard bars descend. Default 5 */
132 struct zint_structapp structapp; /* Structured Append info. Default structapp.count 0 (none) */
133 int warn_level; /* Affects error/warning value returned by Zint API (see WARN_XXX below) */
134 int debug; /* Debugging flags */
135 unsigned char text[256]; /* Human Readable Text (HRT) (if any), UTF-8, NUL-terminated (output only) */
136 int text_length; /* Length of text in bytes (output only) */
137 int rows; /* Number of rows used by the symbol (output only) */
138 int width; /* Width of the generated symbol (output only) */
139 char errtxt[160]; /* Error message if an error or warning occurs, NUL-terminated (output only) */
140 unsigned char *bitmap; /* Stored bitmap image (raster output only) */
141 int bitmap_width; /* Width of bitmap image (raster output only) */
142 int bitmap_height; /* Height of bitmap image (raster output only) */
143 unsigned char *alphamap; /* Array of alpha values used (raster output only) */
144 struct zint_vector *vector; /* Pointer to vector header (vector output only) */
145 unsigned char *memfile; /* Pointer to in-memory file buffer if BARCODE_MEMORY_FILE (output only) */
146 int memfile_size; /* Length of in-memory file buffer (output only) */
147 struct zint_seg *content_segs; /* Pointer to array of content segs if BARCODE_CONTENT_SEGS (output only) */
148 int content_seg_count; /* Number of `content_segs` (output only) */
149 unsigned char encoded_data[200][144]; /* Encoded data (output only). Allows for rows of 1152 modules */
150 float row_height[200]; /* Heights of rows (output only). Allows for 200 row DotCode */
151 };
