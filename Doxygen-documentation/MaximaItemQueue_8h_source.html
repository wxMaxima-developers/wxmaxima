170template <class TagMap>
171std::size_t SplitIntoItems(const wxString &buffer, const TagMap &knownTags,
172 int textCause,
173 std::vector<MaximaItemQueue::Item> &items,
174 const std::function<bool()> &stopRequested = {}) {
175 const std::size_t length = buffer.length();
176 wxString text;
177 std::size_t pos = 0;
178
179 while (pos < length) {
180 if (stopRequested && stopRequested())
181 break;
182 const std::size_t tagPos = buffer.find(wxS('<'), pos);
183 if (tagPos == wxString::npos) {
184 text.append(buffer, pos, wxString::npos);
185 pos = length;
186 break;
187 }
188 if (tagPos > pos) {
189 text.append(buffer, pos, tagPos - pos);
190 pos = tagPos;
191 continue;
192 }
193
194 auto tag = knownTags.begin();
195 bool mayBeTagStart = false;
196 for (; tag != knownTags.end(); ++tag) {
197 const wxString tagStartName = wxS("<") + tag->first + wxS(">");
198 if (buffer.compare(pos, tagStartName.length(), tagStartName) == 0)
199 break;
200 // The data ends in the middle of this start tag, for example "<mat".
201 if ((length - pos < tagStartName.length()) &&
202 tagStartName.StartsWith(buffer.substr(pos)))
203 mayBeTagStart = true;
204 }
205 if ((tag == knownTags.end()) && mayBeTagStart)
206 // Decide once the rest has arrived. Taking the "<" for text here used
207 // to turn the whole tag into text: a line of raw XML in the worksheet.
208 break;
209 if (tag == knownTags.end()) {
210 // A "<" that doesn't start a known tag is just text
211 text += buffer[pos];
212 pos++;
213 continue;
214 }
215
216 const wxString tagEndName = wxS("</") + tag->first + wxS(">");
217 const std::size_t end = buffer.find(tagEndName, pos);
218 if (end == wxString::npos)
219 // Not complete yet: wait for the rest of it.
220 break;
221 if (!text.IsEmpty()) {
222 items.push_back({textCause, text});
223 text.Clear();
224 }
225 const std::size_t tagFullLength = end + tagEndName.length() - pos;
226 items.push_back({static_cast<int>(tag->second), buffer.substr(pos, tagFullLength)});
227 pos += tagFullLength;
228 if ((pos < length) && (buffer[pos] == wxS('\n')))
229 pos++;
230 }
231
232 if (!text.IsEmpty())
233 items.push_back({textCause, std::move(text)});
234 return pos;
235}
