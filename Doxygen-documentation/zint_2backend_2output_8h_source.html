81/* Little-endian output */
82#define zint_out_le_u16(b, n) do { \
83 unsigned char *bp = (unsigned char *) &(b); \
84 uint16_t u16 = (uint16_t) (n); \
85 bp[0] = (unsigned char) (u16 & 0xFF); \
86 bp[1] = (unsigned char) ((u16 >> 8) & 0xFF); \
87 } while (0)
88
89#define zint_out_le_u32(b, n) do { \
90 unsigned char *bp = (unsigned char *) &(b); \
91 uint32_t u32 = (uint32_t) (n); \
92 bp[0] = (unsigned char) (u32 & 0xFF); \
93 bp[1] = (unsigned char) ((u32 >> 8) & 0xFF); \
94 bp[2] = (unsigned char) ((u32 >> 16) & 0xFF); \
95 bp[3] = (unsigned char) ((u32 >> 24) & 0xFF); \
96 } while (0)
97
98#define zint_out_le_i32(b, n) do { \
99 unsigned char *bp = (unsigned char *) &(b); \
100 int32_t i32 = (int32_t) (n); \
101 bp[0] = (unsigned char) (i32 & 0xFF); \
102 bp[1] = (unsigned char) ((i32 >> 8) & 0xFF); \
103 bp[2] = (unsigned char) ((i32 >> 16) & 0xFF); \
104 bp[3] = (unsigned char) ((i32 >> 24) & 0xFF); \
105 } while (0)
106
107#define zint_out_le_float(b, n) do { \
108 unsigned char *bp = (unsigned char *) &(b); \
109 float f = (float) (n); \
110 uint32_t *p_u32 = (uint32_t *) &f; \
111 bp[0] = (unsigned char) (*p_u32 & 0xFF); \
112 bp[1] = (unsigned char) ((*p_u32 >> 8) & 0xFF); \
113 bp[2] = (unsigned char) ((*p_u32 >> 16) & 0xFF); \
114 bp[3] = (unsigned char) ((*p_u32 >> 24) & 0xFF); \
115 } while (0)
