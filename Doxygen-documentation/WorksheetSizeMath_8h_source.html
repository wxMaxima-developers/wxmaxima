89inline int ComputeWorksheetContentHeight(
90 const std::vector<TrailingGroupGeometry> &trailing, int groupSkip,
91 int baseIndent) {
92 int extraHeight = 0;
93 for (const TrailingGroupGeometry &cell : trailing) {
94 if (cell.sizeIsStale && cell.currentY >= 0)
95 // The anchor: its top position is trusted; add its drop and everything
96 // accumulated below it.
97 return cell.currentY + cell.maxDrop + extraHeight;
98 if (cell.sizeIsStale)
99 // Stale and not yet positioned (currentY < 0): count its own extent.
100 extraHeight += cell.maxDrop + groupSkip;
101 else
102 // Not stale but still unpositioned mid-layout: reserve a nominal height.
103 extraHeight += groupSkip + 20;
104 }
105 return baseIndent + extraHeight;
106}

138inline WorksheetVirtualSize ComputeWorksheetVirtualSize(bool hasTree,
139 int maxPointWidth,
140 int maxPointHeight,
141 int clientHeight,
142 int currentScrollPixelY) {
143 WorksheetVirtualSize result;
144 int scrollUnitBasisHeight = 40;
145 if (hasTree) {
146 result.width = maxPointWidth;
147 // Allow scrolling a little past the end of the document.
148 int height = maxPointHeight + clientHeight - clientHeight / 8;
149 // Keep a vertical scrollbar active.
150 result.height = std::max(clientHeight + 10, height);
151 // Never shrink below what the current scroll position requires.
152 result.height =
153 std::max(result.height, currentScrollPixelY + clientHeight + 10);
154 scrollUnitBasisHeight = clientHeight;
155 }
156 result.scrollUnit = std::max(scrollUnitBasisHeight / 30, 10);
157 return result;
158}

194inline ScrollCompensation ComputeScrollCompensation(int scrollUnitsY,
195 int scrollUnit,
196 int topOffset, int shift) {
197 ScrollCompensation result;
198 if (scrollUnit < 1)
199 scrollUnit = 1;
200 // The pixel position the view's top edge must be at, measured in a document
201 // whose top offset is 0.
202 const int target = scrollUnitsY * scrollUnit + shift - topOffset;
203 if (target <= 0) {
204 result.scrollUnitsY = 0;
205 result.topOffset = std::clamp(-target, 0, scrollUnit - 1);
206 return result;
207 }
208 // Round up to a whole scroll unit and shift the document down by the excess.
209 result.scrollUnitsY = (target + scrollUnit - 1) / scrollUnit;
210 result.topOffset = result.scrollUnitsY * scrollUnit - target;
211 return result;
212}

266inline void ApplyWorksheetVirtualSize(WorksheetView &view, bool hasTree,
267 int maxWidth, int maxHeight,
268 WorksheetVirtualSizeCache &cache,
269 int &scrollUnit) {
270 int clientWidth, clientHeight;
271 view.GetViewClientSize(&clientWidth, &clientHeight);
272
273 int currentScrollPixelY = 0;
274 if (hasTree)
275 currentScrollPixelY = view.GetViewScrollUnitY() * scrollUnit;
276
277 const WorksheetVirtualSize vs = ComputeWorksheetVirtualSize(
278 hasTree, maxWidth, maxHeight, clientHeight, currentScrollPixelY);
279
280 if ((cache.lastWidth != vs.width || cache.lastHeight != vs.height) &&
281 vs.height > 0) {
282 cache.lastWidth = vs.width;
283 cache.lastHeight = vs.height;
284 view.SetViewVirtualSize(vs.width, vs.height);
285 scrollUnit = vs.scrollUnit;
286 view.SetViewScrollRate(vs.scrollUnit);
287 }
288}
