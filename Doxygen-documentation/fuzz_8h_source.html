123/* Verifies if a string only uses valid characters, returning 1-based position in `source` if not, 0 for success */
124static int not_sane(const unsigned int flg, const unsigned char source[], const int length) {
125 int i;
126
127 for (i = 0; i < length; i++) {
128 if (!(flgs[source[i]] & flg)) {
129 return i + 1;
130 }
131 }
132 return 0;
133}

324/* Set `symbol` based on `settings[idx]`, returning new length after consuming some `*p_input` (`data`) */
325static int set_symbol(struct zint_symbol *symbol, const int idx, const int chk_sane, const int no_eci,
326 const unsigned char **p_input, const size_t size) {
327 const unsigned char *input = *p_input;
328 int length = (int) size;
329 const struct settings_item *si;
330
331 assert(idx >= 0 && idx < ZARRAY_SIZE(settings));
332 assert(settings[idx].symbology > 0);
333 assert(settings[idx].symbology == idx);
334
335 si = settings + idx;
336
337 ZBarcode_Reset(symbol);
338
339 symbol->symbology = si->symbology;
340 if (length < si->len_min) {
341 return 0;
342 }
343 /* `input_mode` */
344 if (length > si->len_min) {
345 symbol->input_mode = (*input++ << 3); /* Note DATA/UNICODE/GS1_MODE must be set by caller */
346 length--;
347 }
348 /* `option_1` */
349 if (length > si->len_min && si->option_1_min <= si->option_1_max) {
350 unsigned char ch = *input++;
351 if (ch != 0xFF) { /* Special case 255 as default (-1) */
352 if (si->option_1_min + 1 == si->option_1_max) { /* Only one in it? */
353 symbol->option_1 = (ch & 1) ? si->option_1_min : si->option_1_max; /* Odd/even */
354 } else {
355 symbol->option_1 = clamp(ch, si->option_1_min, si->option_1_max);
356 }
357 }
358 length--;
359 }
360 /* `option_2` */
361 if (length > si->len_min && si->option_2_min <= si->option_2_max) {
362 if (si->option_2_min + 1 == si->option_2_max) { /* Only one in it? */
363 symbol->option_2 = (*input++ & 1) ? si->option_2_min : si->option_2_max; /* Odd/even */
364 } else {
365 symbol->option_2 = clamp(*input++, si->option_2_min, si->option_2_max);
366 }
367 length--;
368 }
369 /* `eci` */
370 if (!no_eci) {
371 if (length > si->len_min && (ZBarcode_Cap(symbol->symbology, ZINT_CAP_ECI) & ZINT_CAP_ECI)) {
372 symbol->eci = *input++;
373 /* Avoid invalid/unconvertible ECIs */
374 if (symbol->eci == 1) {
375 symbol->eci = 899; /* Binary */
376 } else if (symbol->eci > 35 && symbol->eci != 170) {
377 symbol->eci = 35; /* UTF-32LE */
378 } else if (symbol->eci == 2 || symbol->eci == 15 || symbol->eci == 19) {
379 symbol->eci++;
380 }
381 length--;
382 }
383 }
384 /* `option_3` */
385 if (length > si->len_min && (si->option_3_min <= si->option_3_max)) {
386 if (si->option_3_min + 1 == si->option_3_max) { /* Only one in it? */
387 symbol->option_3 = (*input++ & 1) ? si->option_3_min : si->option_3_max; /* Odd/even */
388 } else {
389 symbol->option_3 = clamp(*input++, si->option_3_min, si->option_3_max);
390 }
391 length--;
392 }
393 /* `height` */
394 if (length > si->len_min && !(ZBarcode_Cap(symbol->symbology, ZINT_CAP_FIXED_RATIO) & ZINT_CAP_FIXED_RATIO)) {
395 symbol->height = *input++;
396 length--;
397 }
398 /* `border_width` */
399 if (length > si->len_min) {
400 symbol->border_width = *input++;
401 length--;
402 }
403 /* `output_options` */
404 if (length > si->len_min) {
405 symbol->output_options = *input++ & OUTPUT_OPTIONS_MASK;
406 length--;
407 }
408
409 if (length > si->len_max) {
410 return 0;
411 }
412 if (chk_sane && si->sane_flag && not_sane(si->sane_flag, input, length)) {
413 return 0;
414 }
415
416#ifdef Z_FUZZ_DEBUG
417 symbol->debug = ZINT_DEBUG_PRINT;
418#endif
419
420 *p_input = input;
421
422 assert(length >= si->len_min && length <= si->len_max);
423
424 return length;
425}
