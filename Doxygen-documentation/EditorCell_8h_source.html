103 void UpdateSelectionString();
104 // The selection/cursor setters clamp their argument to a valid index into the
105 // current text via ClampToText(), so m_selectionStart / m_selectionEnd are
106 // always in [0, m_text.Length()] (e.g. CursorMove(-n) past the start clamps to
107 // 0 instead of wrapping a size_t to a huge value). The getters below keep a
108 // defensive std::min as well, because the text can still shrink *after* a
109 // selection was set without the selection being updated.
110 void SetSelection(size_t start, size_t end){m_selectionStart = ClampToText(start); m_selectionEnd = ClampToText(end); UpdateSelectionString();}
